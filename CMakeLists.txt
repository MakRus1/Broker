cmake_minimum_required(VERSION 3.24...3.31)
project(broker-monorepo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(STRINGS "${CMAKE_SOURCE_DIR}/tools/versions.env" version_lines REGEX "^[A-Z0-9_]+=")
foreach(line IN LISTS version_lines)
    if(line MATCHES "^([A-Z0-9_]+)=(.*)$")
        set("BROKER_${CMAKE_MATCH_1}" "${CMAKE_MATCH_2}")
    endif()
endforeach()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter Development)
set(USERVER_PYTHON_PATH "${Python3_EXECUTABLE}" CACHE FILEPATH "Интерпретатор testsuite" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
file(SHA256 "${CMAKE_SOURCE_DIR}/tools/python-constraints.txt" python_constraints_hash)
set(python_constraints "${CMAKE_BINARY_DIR}/python-constraints-${python_constraints_hash}.txt")
configure_file("${CMAKE_SOURCE_DIR}/tools/python-constraints.txt" "${python_constraints}" COPYONLY)
set(USERVER_PIP_OPTIONS "--constraint;${python_constraints}" CACHE STRING "Закрепленные зависимости userver" FORCE)

set(BOOST_IOSTREAMS_ENABLE_ZLIB ON CACHE BOOL "Поддержка gzip в userver" FORCE)
set(USERVER_FEATURE_GRPC ON CACHE BOOL "" FORCE)
set(USERVER_FEATURE_POSTGRESQL ON CACHE BOOL "" FORCE)
set(USERVER_FEATURE_CHAOTIC ON CACHE BOOL "" FORCE)
set(USERVER_FEATURE_CHAOTIC_OPENAPI OFF CACHE BOOL "" FORCE)
set(USERVER_FEATURE_OTLP ON CACHE BOOL "" FORCE)
include(NativeDependencies)
# Предсобранный framework должен соответствовать выбранному режиму санитайзеров.
if(DEFINED ENV{BROKER_USERVER_PREFIX})
    set(broker_userver_prefix "$ENV{BROKER_USERVER_PREFIX}")
    file(READ "${broker_userver_prefix}/broker-sanitizers.txt" broker_image_sanitizers)
    string(STRIP "${broker_image_sanitizers}" broker_image_sanitizers)
    if(NOT "${broker_image_sanitizers}" STREQUAL "${USERVER_SANITIZE}")
        message(FATAL_ERROR "Образ userver не соответствует USERVER_SANITIZE; выберите dev или sanitize")
    endif()
    find_package(userver ${BROKER_USERVER_VERSION} EXACT REQUIRED COMPONENTS core grpc postgresql otlp
        PATHS "${broker_userver_prefix}" NO_DEFAULT_PATH)
elseif(NOT USERVER_SANITIZE)
    find_package(userver ${BROKER_USERVER_VERSION} EXACT QUIET COMPONENTS core grpc postgresql otlp)
endif()
if(NOT userver_FOUND)
    include(DownloadUserver)
    download_userver()
endif()
userver_setup_environment()
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import yaml" RESULT_VARIABLE yaml_missing)
if(yaml_missing)
    message(FATAL_ERROR "Install requirements.txt into the Python environment selected by Python3_EXECUTABLE")
endif()
include(CTest)
file(GLOB library_cmake_files CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/libs/*/CMakeLists.txt")
foreach(library_cmake IN LISTS library_cmake_files)
    get_filename_component(library_dir "${library_cmake}" DIRECTORY)
    add_subdirectory("${library_dir}")
endforeach()
add_subdirectory(services/broker)

if(BUILD_TESTING)
    add_subdirectory(tests/fixtures/contract_service)
endif()

if(APPLE)
    cmake_language(DEFER CALL broker_native_headers "${CMAKE_SOURCE_DIR}")
endif()
